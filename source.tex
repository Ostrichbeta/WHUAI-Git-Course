    \begin{center}

        \item 暫存區：準備納入下一次 commit 的檔案內容。
        \item commit：把暫存區的專案狀態保存成一個版本。
    \end{itemize}
\end{frame}

pwd

\begin{frame}{遇到問題，先讀 git status}
\begin{block}{讀懂三種常見狀態}
        \vspace{0.5em}
        \setlength{\parskip}{0pt}
        \begin{tabular}{@{}l@{\hspace{1em}}l@{}}
            \strut\textcolor{maincolor}{\texttt{Untracked}} &
                \parbox[t]{0.65\textwidth}{\strut Git 還沒開始追蹤這個新檔案，尚未納入版本控制。}\\[0.7em]
            \strut\textcolor{maincolor}{\texttt{Not staged}} &
                \parbox[t]{0.65\textwidth}{\strut 已追蹤的檔案有修改，尚未加入暫存區。}\\[0.7em]
            \strut\textcolor{maincolor}{\texttt{To be committed}} &
                \parbox[t]{0.65\textwidth}{\strut 變更已進暫存區，下次 commit 會包含這些內容。}
        \end{tabular}
    \end{block}
    \begin{itemize}
        \item 同一個檔案可以同時有已 staged 和尚未 staged 的變更。
        \item \texttt{working tree clean}：本機沒有待 commit 的變更，不代表已 push。
    \end{itemize}
\end{frame}

% S09
\begin{frame}[fragile]{保存第一個版本}
\begin{lstlisting}[language=bash]
git add README.md grades.py
git status
git commit -m "Add grade calculator"
git status
\end{lstlisting}
    \begin{itemize}
        \item \texttt{add}：選擇這次要保存的內容。
        \item \texttt{commit -m}：保存版本，說明這次改了什麼。
        \item commit message 格式沒有強制規定，但內容要精確、簡潔。
    \end{itemize}
\end{frame}

% S10
\begin{frame}[fragile]{先 add，再修改：哪一版會被 commit？}
    \begin{enumerate}
        \item 在 README 加入 \texttt{Input: a list of scores.}
        \item 執行 \texttt{git add README.md}。
        \item 再加入 \texttt{Output: the average score.}，先不要 add。
    \end{enumerate}
\begin{lstlisting}[language=bash]
git status
git diff
git diff --staged
\end{lstlisting}
    \begin{itemize}
        \item \texttt{diff}：工作目錄與暫存區的差異。
        \item \texttt{diff --staged}：暫存區與上次 commit 的差異。
    \end{itemize}
\end{frame}

% S11
\begin{frame}{練習 1：把兩次修改分開保存}
    \begin{enumerate}
        \item 承接上一頁，只 commit Input 那一行。
        \item 確認 Output 那一行還留在工作目錄。
        \item 將 Output 那一行留到下一個 commit。
    \end{enumerate}
    \begin{block}{完成條件}
        \begin{itemize}
            \item 至少三個 commit：初始專案、輸入說明、輸出說明。
            \item 工作目錄乾淨，並能說明兩次 diff 各自比較的對象。
        \end{itemize}
    \end{block}
\end{frame}

    \end{columns}

\end{frame}

% S13
\begin{frame}[fragile]{用歷史回答：改了什麼？為什麼？}
\begin{lstlisting}[language=bash]
git log --oneline
git show HEAD
\end{lstlisting}
    \begin{itemize}
        \item \texttt{--oneline} 讓每個 commit 只佔一行，顯示縮短的 commit ID 與 message 首行。
        \item \texttt{HEAD} 通常經由目前 branch 指向該 branch 的最新 commit。
        \item \texttt{show HEAD} 可以檢查最近一次 commit 的內容。
        \item 出現分頁畫面時按 \texttt{q} 返回終端機。
    \end{itemize}
\end{frame}

\section{branch 與合併}

% S16
\begin{frame}{branch 代表另一條開發路線}
\begin{center}
    \begin{tikzpicture}[node distance=2.8cm]
        \node[commit] (a) {A};
        \node[commit,right of=a] (b) {B};
        \node[activecommit,right of=b] (c) {C};
        \draw[arrow] (b) -- (a);
        \draw[arrow] (c) -- (b);
        \node[above=0.85cm of a,font=\footnotesize] (old) {Older};
        \node[above=0.85cm of c,font=\footnotesize] (new) {Newer};
        \draw[->,dashed,maincolor] (old) -- node[above,font=\footnotesize] {Time} (new);
        \node[below=0.4cm of b,font=\small\ttfamily] (main) {main};
        \node[below=0.4cm of c,font=\small\ttfamily] (feature) {handle-empty};
        \draw[arrow] (main) -- (b);
        \draw[arrow] (feature) -- (c);
        \node[below=1.05cm of b,font=\scriptsize] {Commit arrows point to parents};
    \end{tikzpicture}
    \end{center}
    \begin{itemize}
        \item branch 名稱指向某個 commit；產生新 commit 時，目前 branch 會往前移動。
        \item 圖中功能 branch 多出 commit C，\texttt{main} 仍停在 B。
        \item \texttt{HEAD} 通常指向目前所在的 branch。
    \end{itemize}
\end{frame}

% S17
\begin{frame}[fragile]{在功能 branch 處理空的分數清單}
\begin{lstlisting}[language=bash]
git switch -c handle-empty
\end{lstlisting}

\begin{lstlisting}[language=Python]
def average(scores):
    if not scores:
        return 0
    return sum(scores) / len(scores)
\end{lstlisting}
\begin{lstlisting}[language=bash]
git add grades.py
git commit -m "Handle empty score lists"
\end{lstlisting}
\end{frame}

        \draw[arrow] (b) -- (a);
        \draw[arrow] (c) -- (a);
        \node[left=0.18cm of a] {Older};
        \node[right=0.2cm of b,font=\small\ttfamily] {main};
        \node[right=0.2cm of c,font=\small\ttfamily] {improve-title};
    \end{tikzpicture}
    \end{center}
    \begin{columns}[t]
        \begin{column}{0.47\textwidth}
            \begin{block}{\texttt{improve-title}}
                README 第一行改為：\\
                \texttt{\# Team grade calculator}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{block}{\texttt{main}}
                README 第一行改為：\\
                \texttt{\# Class grade calculator}
            \end{block}
        \end{column}
    \end{columns}
    回到 \texttt{main}，執行 \texttt{git merge improve-title}。
    \pause
    \begin{center}\Emph{Git 無法判斷該保留哪個標題，需要我們自行決定。}\end{center}
\end{frame}

% S21
\begin{frame}[fragile]{讀懂衝突標記，保留雙方的用意}
    開啟發生衝突的 \texttt{README.md}，查看 Git 標出的衝突內容。
\begin{lstlisting}
<<<<<<< HEAD
# Class grade calculator
=======
# Team grade calculator
>>>>>>> improve-title
\end{lstlisting}
    \begin{itemize}
        \item 上半部是目前 \texttt{main} 的內容，下半部是要併入的 branch 內容。
        \item 第一行改為 \texttt{\# Team grade calculator for class}。
        \item 刪除全部衝突標記，保留其餘說明。
    \end{itemize}
\end{frame}

% S22
\begin{frame}[fragile]{解決後，還要標記並完成 commit}
\begin{lstlisting}[language=bash]
git status
git add README.md
git commit -m "Combine team and class descriptions"
git status
\end{lstlisting}
    \begin{itemize}
        \item 編輯檔案並檢查結果。
        \item \texttt{add} 告訴 Git：這個檔案的衝突已處理。
        \item \texttt{commit} 完成合併；之後再檢查與驗證。
        \item 取消尚未完成的合併：\texttt{git merge --abort}。
    \end{itemize}
\end{frame}

% S23
\begin{frame}{練習 4：完成一次有衝突的合併}
    \begin{enumerate}
        \item 依講義建立兩個修改標題的 commit，重現衝突。
        \item 整合雙方意思，暫存並完成merge commit。
        \item 用 \texttt{git log --oneline --graph --all} 檢查歷史。
    \end{enumerate}
    \begin{block}{完成條件}
        \begin{itemize}
            \item \texttt{main} 的標題保留 Team 與 class，其餘說明也要保留。
            \item 沒有衝突標記，工作目錄乾淨。
            \item 能解釋編輯後為何還要 add 與 commit。
        \end{itemize}
    \end{block}
\end{frame}

\section{分享與協作}

% S24
\begin{frame}{遠端是另一份倉庫；origin 是它的別名}
\setlength{\parskip}{0pt}
    \begin{tabular}{@{}l@{\hspace{1em}}l@{}}
        \strut\textcolor{maincolor}{\texttt{main}} &
            \parbox[t]{0.70\textwidth}{\strut 自己電腦上的 branch。}\\[0.7em]
        \strut\textcolor{maincolor}{\texttt{origin}} &
            \parbox[t]{0.70\textwidth}{\strut 本機給遠端網址取的別名。}\\[0.7em]
        \strut\textcolor{maincolor}{\texttt{origin/main}} &
            \parbox[t]{0.70\textwidth}{\strut 本機記錄的遠端 \texttt{main} 狀態，可能尚未更新。}
    \end{tabular}
    \begin{itemize}
        \item \texttt{push}：把本機 commit 送到遠端。
        \item \texttt{fetch}：取得遠端 commit，更新本機記錄的遠端狀態。
        \item fetch 不會直接更新目前工作目錄的檔案。
    \end{itemize}
\end{frame}

% S25
\begin{frame}[fragile]{把自己的練習倉庫推到 GitHub}
\small
先在 GitHub 建立 \texttt{git-grade-lab} 空倉庫，先不新增 README、授權或 .gitignore。
    \begin{center}
        \includegraphics[width=0.92\textwidth,trim=0 24bp 0 396bp,clip]{img/github-new-repo-tips.png}
    \end{center}
    \begin{itemize}
        \item 本機已有 repository，請看 GitHub 空倉庫頁的 push an existing repository 區塊。
        \item 圖片是範例，請複製自己 repository 的 URL，不要照抄圖片中的帳號。
    \end{itemize}
\begin{lstlisting}[language=bash]
git remote add origin https://github.com/YOUR-USER/git-grade-lab.git
git push -u origin main
\end{lstlisting}
    {\footnotesize 課堂已使用 \texttt{main}，GitHub 指示的 \texttt{git branch -M main} 不必重做。}
\end{frame}

% S26
\begin{frame}[fragile]{取得既有專案，使用 clone}
\begin{lstlisting}[language=bash]
git clone https://github.com/YOUR-USER/git-grade-lab.git git-grade-copy
cd git-grade-copy
git status
\end{lstlisting}
    \begin{itemize}
        \item clone 會取得倉庫，並設定 origin 與預設 branch 的上游。
        \item 複製好之後不需要再 init。
        \item 回到\Emph{原本練習目錄}繼續下一段。
    \end{itemize}
    只有遠端有新增 commit、自己沒有額外 commit 時，可以：
\begin{lstlisting}[language=bash]
git pull --ff-only
\end{lstlisting}
\end{frame}

% S27
\begin{frame}[fragile]{push 被拒絕：先整合遠端更新}
    情境：在 \texttt{main}，兩邊各有對方沒有的 commit，工作目錄乾淨。
\begin{lstlisting}[language=bash]
git fetch origin
git merge --no-edit origin/main
git push
\end{lstlisting}
    \begin{itemize}
        \item 沒有內容衝突時，Git 可以直接整合。
        \item 若有衝突，依序編輯、add、commit，再 push。
        \item \texttt{--no-edit} 接受自動產生的合併訊息。
    \end{itemize}
    \begin{center}\Emph{兩邊各有新 commit，不一定代表內容衝突。}\end{center}
\end{frame}

% S28
\begin{frame}[fragile]{用 Pull Request 讓同學審閱修改}
\begin{lstlisting}[language=bash]
git switch -c docs-usage
\end{lstlisting}
    在 README 加上使用說明後：
\begin{lstlisting}[language=bash]
git add README.md
git commit -m "Explain how to run the calculator"
git push -u origin docs-usage
\end{lstlisting}
    \begin{itemize}
        \item 在 GitHub 開 PR：base 選 \texttt{main}，compare 選 \texttt{docs-usage}。
        \item 說明目的、內容與檢查方式，請同學審閱。
        \item 修改後繼續 commit 並 push 到同一 branch，原本的 PR 就會更新。
    \end{itemize}
\end{frame}

% S29
\begin{frame}[fragile]{練習 5：兩人完成一次審閱與合併}
    \begin{enumerate}
        \item 各自在自己的倉庫提出 \texttt{docs-usage} 的 PR。
        \item 交換連結，檢查用法是否清楚，留下具體建議。
        \item 作者回應並更新同一個 PR，再由有權限的人合併。
        \item 回到本機 \texttt{main}，取得合併後的版本。
    \end{enumerate}
\begin{lstlisting}[language=bash]
git switch main
git pull --ff-only
\end{lstlisting}
    \begin{block}{完成條件}
        PR 有說明與審閱紀錄；本機 \texttt{main} 有使用說明，工作目錄乾淨。
    \end{block}
\end{frame}

% A02
\begin{frame}{先把基本流程練熟，再學進階操作}
    \begin{itemize}
        \item \texttt{stash}：暫時收起尚未 commit 的工作。
        \item \texttt{rebase / cherry-pick}：重整或搬移 commit。
        \item \texttt{reset / reflog}：移動 branch、查找本機操作紀錄。
    \end{itemize}
    \begin{block}{恢復能力有邊界}
        \begin{itemize}
            \item Git 沒有保存過的內容，不能保證找回。
            \item hard reset 會覆蓋工作目錄內容，本次練習不需要用到它。
            \item 改寫已分享的歷史前，先與團隊確認工作方式。
        \end{itemize}
    \end{block}
\end{frame}
